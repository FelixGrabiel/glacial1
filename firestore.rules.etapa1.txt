// =====================================================================
// GLACIAL · REGLAS DE FIRESTORE — ETAPA 1 (compatibles con el sistema actual)
// =====================================================================
// Cómo aplicarlas:
//   Firebase Console > Firestore Database > Reglas > pegar > Publicar.
//
// Cambios respecto a las reglas actuales:
//   - Fase C de Mantenimiento: colección bitacoraMantenimiento (solo se agrega).
//   - Bloqueo de tareos: 'configTareo' y las colecciones relojServidor y solicitudesCorreccionTareo.
//   - Se agregan 'tecnicosMant' y 'borradoresNuevoRegistro' (antes faltaban), 'perfiles' (nuevo: uid -> rol), 'accesos' (nuevo: usuario -> correo interno), 'rotacionesMantenimiento'
//     y 'rotacionMaquinistas' a la lista de documentos permitidos.
//   - Todavía NO se exige inicio de sesión (así nada deja de funcionar
//     mientras se migran los usuarios). La etapa 2 lo exige.
// =====================================================================
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /sync/{doc} {
      allow read, write: if doc in [
        'users', 'perfiles', 'accesos', 'records', 'workers', 'rotaciones',
        'rotacionesMantenimiento', 'rotacionMaquinistas', 'tecnicosMant', 'borradoresNuevoRegistro', 'configTareo', 'configMantenimiento', 'configAlertas', 'configIndicadores', 'cuentasAntiguas', 'tareos',
        'precios', 'paletas', 'programaciones', 'avancesTurno'
      ];
    }
    match /auditoriaTareos/{evento} {
      allow read, create: if true;
      allow update, delete: if false;
    }
    // Bloqueo de tareos: hora del servidor y solicitudes de corrección.
    match /relojServidor/{id} {
      allow read, create, update: if true;
      allow delete: if false;
    }
    match /solicitudesCorreccionTareo/{id} {
      allow read, create, update: if true;
      allow delete: if false;
    }
    // Bitácora de acciones sobre líneas (Fase C de Mantenimiento): solo se agrega.
    match /bitacoraMantenimiento/{id} {
      allow read, create: if true;
      allow update, delete: if false;
    }
    // Resumen de turno (Lote 4 · Parte B): un documento por fecha y turno.
    match /resumenesTurno/{id} {
      allow read, create, update: if true;
      allow delete: if false;
    }
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
