// ARCHIVO ÚNICO Y FINAL. Incluye el rol de la cuenta compartida de Mantenimiento
// (antes en firestore.rules.mantenimiento.txt, ya eliminado).
// Firestore solo admite UN conjunto de reglas activo: se publica este archivo
// completo, reemplazando lo que haya.
// =====================================================================
// ⚠ NO publicar hasta que:
//   1. Todos los usuarios estén migrados a cuentas seguras
//      (botón "Migrar usuarios a cuentas seguras") y hayan entrado
//      al menos una vez con su contraseña nueva.
//   2. LOGIN_LEGACY_PERMITIDO esté en false y REGLAS_ESTRICTAS en true (js/nucleo/01-config.js).
//   3. Se haya probado el sistema con las reglas de la etapa 1.
// Si algo falla después de publicar, vuelve a pegar las reglas de la
// etapa 1 (no se pierde ningún dato).
//
// Cómo funcionan:
//   - Solo entra quien tenga sesión de Firebase Authentication Y figure
//     en el documento sync/perfiles (uid -> rol). Quitar a un usuario de
//     la lista de usuarios le quita el acceso.
//   - users, perfiles y accesos solo los escribe el Administrador.
//   - Los datos operativos los escriben usuarios autenticados.
//   - CUENTA COMPARTIDA DE MANTENIMIENTO (rol 'mantenimiento_compartido'):
//       · Lee solo los documentos de lecturaMant() (no precios, no borradores,
//         no rotaciones de Producción). El código tampoco los escucha para ese rol.
//       · FASE C: opera líneas (detener, pausar, intervenir, reanudar). Para eso recibe
//         DOS permisos de escritura y nada más:
//           1) sync/programaciones: puede actualizar el documento, PERO sin agregar ni
//              quitar programaciones (mismo número de items) y sin otros campos.
//              Las reglas NO pueden mirar qué cambió dentro de cada item: la
//              trazabilidad la da la bitácora y el PIN, no una barrera del servidor.
//           2) bitacoraMantenimiento: crear eventos propios (su uid, hora del servidor).
//         Además escribe sync/tecnicosMant (PIN de técnicos). Con Cloud Functions
//         (MANT_PIN_MODO='functions') quitar esa excepción.
//       · No lee ni escribe auditoriaTareos ni las solicitudes de corrección de tareos.
//   - Limitación conocida: estas reglas validan QUIÉN eres, no cada campo
//     que cambias. El control fino por permiso (tareo, paletas, etc.)
//     sigue en la aplicación; endurecerlo por rol es una etapa posterior.
// =====================================================================
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function perfiles() {
      return get(/databases/$(database)/documents/sync/perfiles).data.map;
    }
    function autenticado() {
      return request.auth != null && request.auth.uid in perfiles();
    }
    function esAdmin() {
      return autenticado() && perfiles()[request.auth.uid].rol == 'Administrador';
    }
    // MODO VISUALIZACIÓN GENERAL (por usuario): el Administrador marca la cuenta como solo consulta y la aplicación publica
    // soloVista:true en su entrada de sync/perfiles (js/accesos/36-seguridad-auth.js). Una cuenta así NO escribe nada operativo,
    // aunque conserve permisos guardados: todas las funciones de abajo que conceden escritura la excluyen.
    // Limitación: estas reglas protegen la ESCRITURA; la lectura por módulo o por línea no se puede imponer porque los
    // documentos compartidos (records, paletas, programaciones…) mezclan todas las líneas (el filtro por línea es del navegador).
    function soloVista() {
      return autenticado() && perfiles()[request.auth.uid].get('soloVista', false) == true;
    }
    function esMantCompartido() {
      return autenticado() && perfiles()[request.auth.uid].rol == 'mantenimiento_compartido';
    }
    // RRHH = permiso moduloRRHH (o Administrador) y NO un rol de solo consulta
    // (Jefatura/Gerencia tienen moduloRRHH solo para ver).
    function esRRHH() {
      return autenticado() && !soloVista() && !(perfiles()[request.auth.uid].rol in
        ['Jefe de Producción', 'Jefe de Operaciones', 'Jefatura', 'Gerente General', 'Gerente']) && (
          esAdmin() ||
          (perfiles()[request.auth.uid].permisos is list &&
           'moduloRRHH' in perfiles()[request.auth.uid].permisos)
        );
    }
    // Supervisor de Mantenimiento (o Administrador): configura la duración de la identificación de técnicos.
    // Las reglas no distinguen el cargo exacto: se exige el permiso de Mantenimiento y que no sea un rol de solo consulta.
    function esSupervisorMtto() {
      return autenticado() && !esMantCompartido() && !soloVista()
        && !(perfiles()[request.auth.uid].rol in
          ['Jefe de Producción', 'Jefe de Operaciones', 'Jefatura', 'Gerente General', 'Gerente'])
        && (esAdmin() || (perfiles()[request.auth.uid].permisos is list
            && ('moduloMantenimiento' in perfiles()[request.auth.uid].permisos
                || 'gestionar_rotacion_mantenimiento' in perfiles()[request.auth.uid].permisos)));
    }
    // Pantalla «Bitácora de Mantenimiento» (solo lectura): Administrador, Jefatura/Gerencia,
    // Supervisor de Producción y Supervisor de Mantenimiento (permiso gestionar_rotacion_mantenimiento).
    // La cuenta compartida de Mantenimiento NO la lee (solo puede crear sus propios eventos).
    // Misma lista que ROLES_LECTURA en js/mantenimiento/43-bitacora-mantenimiento.js.
    function puedeLeerBitacoraMtto() {
      return autenticado() && !esMantCompartido() && (
        esAdmin() ||
        perfiles()[request.auth.uid].rol in ['Jefe de Producción', 'Jefe de Operaciones', 'Jefatura',
          'Gerente General', 'Gerente', 'Supervisor', 'Supervisor de Producción'] ||
        (perfiles()[request.auth.uid].permisos is list
          && ('gestionar_rotacion_mantenimiento' in perfiles()[request.auth.uid].permisos
              || 'ver_bitacora_mantenimiento' in perfiles()[request.auth.uid].permisos))
      );
    }
    // Jefatura/Gerencia (solo para editar los umbrales de avisos y proyección, sync/configAlertas).
    function esJefatura() {
      return autenticado() && perfiles()[request.auth.uid].rol in
        ['Jefe de Producción', 'Jefe de Operaciones', 'Jefatura', 'Gerente General', 'Gerente'];
    }
    // Quién puede editar los umbrales (sync/configAlertas) y las metas/velocidades (sync/configIndicadores):
    // Administrador, Jefatura/Gerencia o quien tenga el permiso configurar_umbrales.
    function puedeConfigurarUmbrales() {
      return esAdmin() || (!soloVista() && (esJefatura() ||
        (autenticado() && perfiles()[request.auth.uid].permisos is list
          && 'configurar_umbrales' in perfiles()[request.auth.uid].permisos)));
    }
    // Quién puede registrar en la bitácora las acciones de PRODUCCIÓN sobre una programación/línea
    // (INICIAR, FINALIZAR, CANCELAR, REABRIR, CORREGIR_INICIO, CORREGIR_FIN): Administrador, Jefe de Producción /
    // de Operaciones, Supervisor (de Producción) o quien tenga el permiso de paletas. NO la cuenta compartida.
    function puedeOperarProduccion() {
      return autenticado() && !esMantCompartido() && !soloVista() && (
        esAdmin() ||
        perfiles()[request.auth.uid].rol in ['Jefe de Producción', 'Jefe de Operaciones', 'Supervisor', 'Supervisor de Producción'] ||
        (perfiles()[request.auth.uid].permisos is list && 'paletas' in perfiles()[request.auth.uid].permisos)
      );
    }
    // Quién guarda o actualiza el Resumen de turno (colección resumenesTurno): quien opera Producción
    // o tiene el permiso avanceProduccion (el mismo que genera el cierre de turno). NO la cuenta compartida.
    function puedeGenerarResumen() {
      return puedeOperarProduccion() ||
        (autenticado() && !esMantCompartido() && !soloVista() && perfiles()[request.auth.uid].permisos is list
          && 'avanceProduccion' in perfiles()[request.auth.uid].permisos);
    }
    // Planificación: permiso «planificacion». Por defecto lo tienen el Administrador, el Jefe de Producción y los roles
    // Planificación y Ventas y Planificación (misma lista que ROLES_PLANIFICACION en js/nucleo/02-estado.js);
    // el Administrador lo da o lo quita en Gestión de usuarios ('planificacion' / '-planificacion'). NO la cuenta compartida.
    function puedePlanificar() {
      return autenticado() && !esMantCompartido() && !soloVista() && (
        esAdmin() ||
        (perfiles()[request.auth.uid].permisos is list && 'planificacion' in perfiles()[request.auth.uid].permisos) ||
        (perfiles()[request.auth.uid].rol in ['Jefe de Producción', 'Planificación', 'Ventas y Planificación']
          && !(perfiles()[request.auth.uid].permisos is list && '-planificacion' in perfiles()[request.auth.uid].permisos))
      );
    }
    // Distribución de personal por línea (Tareo de Producción): Administrador, Supervisor y Jefe de Producción por defecto
    // (misma lista que ROLES_DISTRIBUCION_PERSONAL en js/nucleo/02-estado.js); el Administrador lo da o lo quita
    // ('distribuirPersonal' / '-distribuirPersonal'). NO la cuenta compartida. Jefatura/Gerencia solo consultan.
    function puedeDistribuirPersonal() {
      return autenticado() && !esMantCompartido() && !soloVista() && (
        esAdmin() ||
        (perfiles()[request.auth.uid].permisos is list && 'distribuirPersonal' in perfiles()[request.auth.uid].permisos) ||
        (perfiles()[request.auth.uid].rol in ['Supervisor', 'Jefe de Producción']
          && !(perfiles()[request.auth.uid].permisos is list && '-distribuirPersonal' in perfiles()[request.auth.uid].permisos))
      );
    }
    // INFORMACIÓN ECONÓMICA SENSIBLE: la decide un PERMISO EXPLÍCITO por usuario (no el rol por sí solo ni «todos»).
    //   sync/perfiles[uid].eco = {ver: bool, gestionar: bool}  — lo publica 36-seguridad-auth.js a partir de users[].economico.
    //   El rol solo define quién es ELEGIBLE (se vuelve a comprobar aquí):
    //     ver (consulta de montos en soles)        → Gerencia, Jefatura y Administrador (este último de forma temporal, con permiso explícito)
    //     gestionar (valores unitarios y publicación) → Gerencia y Administrador (temporal, con permiso explícito)
    //   Gerencia conserva su acceso anterior mientras Administración no le asigne permisos explícitos (js/produccion/55-valores-economicos.js).
    //   Jefatura sin permiso explícito NO lee resultados monetarios. Misma lista de roles que 55-valores-economicos.js.
    function ecoGestionar() {
      return autenticado() && !esMantCompartido()
        && perfiles()[request.auth.uid].rol in ['Gerente General', 'Gerente', 'Administrador']
        && perfiles()[request.auth.uid].get('eco', {}).get('gestionar', false) == true;
    }
    function ecoVer() {
      return ecoGestionar() || (autenticado() && !esMantCompartido()
        && perfiles()[request.auth.uid].rol in ['Gerente General', 'Gerente', 'Jefe de Producción', 'Jefe de Operaciones', 'Jefatura', 'Administrador']
        && perfiles()[request.auth.uid].get('eco', {}).get('ver', false) == true);
    }
    // Programaciones: sin altas ni bajas de items y sin campos nuevos en el documento.
    function programacionesSinAltasNiBajas() {
      return resource != null
        && request.resource.data.keys().hasOnly(['items', 'updatedAt'])
        && request.resource.data.items is list
        && request.resource.data.items.size() == resource.data.items.size();
    }
    function lecturaMant(doc) {
      return doc in [
        'users', 'workers', 'rotacionesMantenimiento', 'rotacionMaquinistas',
        'tareos', 'programaciones', 'paletas', 'avancesTurno', 'records',
        'tecnicosMant', 'configMantenimiento', 'configAlertas'
      ];
    }

    match /sync/{doc} {
      // 'accesos' (usuario -> correo interno) es público: lo necesita el login ANTES de iniciar sesión.
      // 'perfiles' (uid -> rol) solo lo lee quien ya figura en perfiles: el login y el arranque de la
      // app no lo leen (las reglas lo consultan por su cuenta); solo lo usa el Administrador al
      // comprobar la migración. 'cuentasAntiguas' (correos por borrar en Authentication) solo el Administrador.
      allow read: if doc == 'accesos'
        || (doc == 'cuentasAntiguas' && esAdmin())
        || (doc == 'precios' && ecoGestionar())
        || (doc != 'cuentasAntiguas' && doc != 'precios' && autenticado() && (!esMantCompartido() || lecturaMant(doc)));

      allow write: if
        // Duración de la identificación de técnicos: solo supervisor de Mantenimiento / Administrador.
        (doc == 'configMantenimiento' && esSupervisorMtto()) ||
        // Cuenta compartida: los PIN de técnicos y operar líneas (programaciones, acotado).
        (esMantCompartido() && doc == 'tecnicosMant') ||
        (esMantCompartido() && doc == 'programaciones' && programacionesSinAltasNiBajas()) ||
        (!esMantCompartido() && (
          (doc in ['users', 'perfiles', 'accesos', 'configTareo', 'cuentasAntiguas'] && esAdmin()) ||
          // Umbrales de avisos y proyección: Administrador y Jefatura. La lee todo usuario autenticado (incluida la cuenta compartida).
          (doc == 'configAlertas' && puedeConfigurarUmbrales()) ||
          // Metas de disponibilidad y velocidades estándar (Análisis de paradas): Administrador y Jefatura.
          // La lee todo usuario autenticado salvo la cuenta compartida (no usa esa pantalla).
          (doc == 'configIndicadores' && puedeConfigurarUmbrales()) ||
          // El Catálogo de Planificación puede editar SOLO las velocidades estándar de esa misma tabla.
          (doc == 'configIndicadores' && puedePlanificar() && resource != null
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['velocidades', 'actualizadoPor', 'actualizadoEn'])) ||
          // Catálogo de productos de Planificación (unidades por paleta y si está activo).
          (doc == 'catalogoPlanificacion' && puedePlanificar()) ||
          // Una cuenta en modo visualización (soloVista) no escribe ningún documento operativo.
          (autenticado() && !soloVista() && doc in [
            'records', 'workers', 'rotaciones', 'rotacionesMantenimiento',
            'rotacionMaquinistas', 'tecnicosMant', 'borradoresNuevoRegistro',
            'tareos', 'paletas', 'programaciones', 'avancesTurno'
          ])
        ));
    }

    // Intervenciones de Mantenimiento (órdenes de trabajo): las LEE quien trabaja o supervisa Mantenimiento, incluida la cuenta compartida
    // (los técnicos son quienes las operan), Administrador, Jefatura/Gerencia (consulta) y supervisores.
    function puedeVerMantenimiento() {
      return autenticado() && (
        esMantCompartido() || esAdmin() ||
        perfiles()[request.auth.uid].rol in ['Jefe de Producción', 'Jefe de Operaciones', 'Jefatura',
          'Gerente General', 'Gerente', 'Supervisor', 'Supervisor de Producción'] ||
        (perfiles()[request.auth.uid].permisos is list
          && ('moduloMantenimiento' in perfiles()[request.auth.uid].permisos
              || 'gestionar_rotacion_mantenimiento' in perfiles()[request.auth.uid].permisos
              || 'ver_bitacora_mantenimiento' in perfiles()[request.auth.uid].permisos))
      );
    }
    // Las ESCRIBE la cuenta compartida (el técnico se identifica con PIN en la aplicación: el servidor no valida el PIN, ver limitación de la
    // cuenta compartida) y el Supervisor de Mantenimiento / Administrador. Jefatura/Gerencia y las cuentas de solo consulta NO.
    function puedeEscribirIntervenciones() {
      return autenticado() && !soloVista()
        && !(perfiles()[request.auth.uid].rol in ['Jefe de Producción', 'Jefe de Operaciones', 'Jefatura', 'Gerente General', 'Gerente'])
        && (esMantCompartido() || esSupervisorMtto());
    }

    // Hora del servidor (bloqueo de tareos): cada usuario solo escribe su propio documento
    // y solo con la hora del servidor.
    match /relojServidor/{id} {
      allow read: if autenticado() && id == request.auth.uid;
      allow create, update: if autenticado() && id == request.auth.uid
        && request.resource.data.keys().hasOnly(['t'])
        && request.resource.data.t == request.time;
      allow delete: if false;
    }

    // Solicitudes de corrección de tareos bloqueados.
    //  - Cualquier usuario con sesión (menos la cuenta compartida) crea la suya, con su uid
    //    y la hora del servidor, siempre como 'Pendiente'.
    //  - Cada quien lee las suyas; RRHH lee todas y solo puede marcarlas como atendidas.
    //  - Nadie las borra.
    match /solicitudesCorreccionTareo/{id} {
      allow read: if autenticado() && !esMantCompartido()
        && (esRRHH() || resource.data.uid == request.auth.uid);
      allow create: if autenticado() && !esMantCompartido() && !soloVista()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.estado == 'Pendiente'
        && request.resource.data.creadoEn == request.time;
      allow update: if esRRHH()
        && request.resource.data.estado == 'Atendida'
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['estado', 'atendidaPor', 'atendidaPorNombre', 'atendidaEn', 'nota']);
      allow delete: if false;
    }

    // Bitácora de acciones sobre líneas (Fase C): solo se agrega, nunca se edita ni se borra.
    // Cada evento lleva el uid de quien lo escribe y la hora del servidor.
    match /bitacoraMantenimiento/{id} {
      allow read: if puedeLeerBitacoraMtto();
      allow create: if autenticado() && !soloVista()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.timestamp == request.time
        && request.resource.data.keys().hasAll(['uid', 'timestamp', 'accion', 'linea', 'tecnicoNombre'])
        && request.resource.data.accion in ['DETENER', 'PAUSAR', 'INTERVENIR', 'REANUDAR', 'COMPLETAR_MOTIVO',
             'INICIAR', 'FINALIZAR', 'CANCELAR', 'REABRIR', 'CORREGIR_INICIO', 'CORREGIR_FIN']
        // Acciones de Producción: solo quien opera Producción y con origen PRODUCCION.
        && (!(request.resource.data.accion in ['INICIAR', 'FINALIZAR', 'CANCELAR', 'REABRIR', 'CORREGIR_INICIO', 'CORREGIR_FIN'])
            || (puedeOperarProduccion() && request.resource.data.origen == 'PRODUCCION'))
        // Correcciones: motivo obligatorio (mínimo 5 caracteres) y valor anterior y nuevo.
        && (!(request.resource.data.accion in ['CORREGIR_INICIO', 'CORREGIR_FIN'])
            || (request.resource.data.motivoCorreccion is string && request.resource.data.motivoCorreccion.size() >= 5
                && request.resource.data.keys().hasAll(['valorAnterior', 'valorNuevo'])))
        // Origen (campo nuevo): si viene, solo PRODUCCION o MANTENIMIENTO.
        && (!('origen' in request.resource.data.keys()) || request.resource.data.origen in ['PRODUCCION', 'MANTENIMIENTO'])
        // COMPLETAR_MOTIVO: evento nuevo (la bitácora no se edita) con la descripción del motivo (mínimo 5 caracteres).
        && (request.resource.data.accion != 'COMPLETAR_MOTIVO'
            || (request.resource.data.motivoDetalle is string && request.resource.data.motivoDetalle.size() >= 5));
      allow update, delete: if false;
    }

    // Resumen de turno (Lote 4 · Parte B): UN documento por fecha y turno (id AAAA-MM-DD_DIA o _NOCHE).
    //  - Lo leen los mismos roles que la bitácora (jefatura, Administrador, supervisores).
    //  - Se crea una sola vez (si ya existe, un segundo cierre no puede pisarlo) con la hora del servidor.
    //  - «Actualizar resumen» solo cambia los datos y quién/cuándo; nunca se borra.
    match /resumenesTurno/{id} {
      allow read: if puedeLeerBitacoraMtto();
      allow create: if puedeGenerarResumen()
        && request.resource.data.keys().hasAll(['fecha', 'turno', 'estado', 'datos', 'creadoUid', 'creadoEn'])
        && request.resource.data.keys().hasOnly(['fecha', 'turno', 'estado', 'datos', 'creadoUid', 'creadoPor',
             'creadoPorNombre', 'creadoEn', 'actualizaciones'])
        && request.resource.data.turno in ['DIA', 'NOCHE']
        && id == request.resource.data.fecha + '_' + request.resource.data.turno
        && request.resource.data.estado == 'CERRADO'
        && request.resource.data.creadoUid == request.auth.uid
        && request.resource.data.creadoEn == request.time;
      allow update: if puedeGenerarResumen()
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['datos', 'actualizadoUid', 'actualizadoPor', 'actualizadoPorNombre', 'actualizadoEn', 'actualizaciones'])
        && request.resource.data.actualizadoUid == request.auth.uid
        && request.resource.data.actualizadoEn == request.time;
      allow delete: if false;
    }

    // Intervenciones de Mantenimiento: UN documento por intervención (id = el id de su origen si nace de una parada/evento, o uno propio).
    //  - Crea quien las escribe, con su uid y la hora del servidor, en estado PENDIENTE o EN_INTERVENCION, versión 1 y UNA entrada de historial.
    //  - Cada cambio sube la versión en 1, lleva uid y hora del servidor y AGREGA exactamente una entrada de historial (el historial nunca se acorta).
    //  - Los campos de origen (id, abiertaUid/En/Ms/Por, origen) no se cambian nunca.
    //  - Una intervención CANCELADA no se modifica. Una CERRADA solo admite: seguimiento y evidencias (cualquiera que escriba), o una corrección con
    //    permiso (Supervisor de Mantenimiento / Administrador) que la deja CERRADA. Nadie borra.
    match /intervenciones/{id} {
      allow read: if puedeVerMantenimiento();
      allow create: if puedeEscribirIntervenciones()
        && request.resource.data.keys().hasAll(['id', 'version', 'linea', 'tipo', 'estado', 'abierta', 'problema', 'abiertaUid', 'abiertaEn',
             'abiertaMs', 'historial', 'actualizadaUid', 'actualizadaEn'])
        && request.resource.data.id == id
        && request.resource.data.version == 1
        && request.resource.data.linea in ['PET1', 'PET2', 'B7L', 'C20L', 'B20L']
        && request.resource.data.estado in ['PENDIENTE', 'EN_INTERVENCION']
        && request.resource.data.abierta == true
        && request.resource.data.abiertaUid == request.auth.uid
        && request.resource.data.abiertaEn == request.time
        && request.resource.data.actualizadaUid == request.auth.uid
        && request.resource.data.actualizadaEn == request.time
        && request.resource.data.historial is list && request.resource.data.historial.size() == 1;
      allow update: if puedeEscribirIntervenciones()
        && request.resource.data.version == resource.data.version + 1
        && request.resource.data.actualizadaUid == request.auth.uid
        && request.resource.data.actualizadaEn == request.time
        && request.resource.data.historial is list
        && request.resource.data.historial.size() == resource.data.historial.size() + 1
        && !request.resource.data.diff(resource.data).affectedKeys()
             .hasAny(['id', 'abiertaUid', 'abiertaEn', 'abiertaMs', 'abiertaPor', 'origen'])
        && resource.data.estado != 'CANCELADA'
        && (resource.data.estado != 'CERRADA' || (
              request.resource.data.estado == 'CERRADA' && (
                request.resource.data.diff(resource.data).affectedKeys()
                  .hasOnly(['seguimiento', 'evidencias', 'historial', 'version', 'actualizadaEn', 'actualizadaUid'])
                || esSupervisorMtto())));
      allow delete: if false;
    }

    // Planes de mantenimiento preventivo (por equipo): UN documento por plan. Los lee quien ve Mantenimiento.
    //  - Los DEFINE (crear y cambiar su contenido) el Supervisor de Mantenimiento / Administrador, con rev + 1, su uid y la hora del servidor.
    //  - Quien opera Mantenimiento (incluida la cuenta compartida) solo puede AVANZAR el ciclo: cerrar una ejecución (próxima fecha y última ejecución) o
    //    reprogramar con motivo (arreglo reprogramaciones). No puede cambiar tareas, frecuencia ni responsable.
    //  - Nadie borra.
    match /planesPreventivos/{id} {
      allow read: if puedeVerMantenimiento();
      allow create: if esSupervisorMtto()
        && request.resource.data.keys().hasAll(['id', 'equipoId', 'titulo', 'frecuencia', 'proximaFecha', 'tareas', 'rev', 'version', 'actualizadaUid', 'actualizadaEn'])
        && request.resource.data.id == id
        && request.resource.data.rev == 1
        && request.resource.data.tareas is list && request.resource.data.tareas.size() > 0 && request.resource.data.tareas.size() <= 30
        && request.resource.data.actualizadaUid == request.auth.uid
        && request.resource.data.actualizadaEn == request.time;
      allow update: if request.resource.data.rev == resource.data.rev + 1
        && request.resource.data.actualizadaUid == request.auth.uid
        && request.resource.data.actualizadaEn == request.time
        && (esSupervisorMtto()
            || (puedeEscribirIntervenciones()
                && request.resource.data.diff(resource.data).affectedKeys()
                     .hasOnly(['proximaFecha', 'fechaOriginal', 'ultimaEjecucion', 'reprogramaciones', 'rev', 'actualizadaEn', 'actualizadaUid'])));
      allow delete: if false;
    }

    // Corregir y reasignar paradas: quién puede dejar constancia de una corrección de asignación.
    // Administrador o quien tenga el permiso corregir_paradas; nunca la cuenta compartida de Mantenimiento ni una cuenta en modo consulta.
    function puedeCorregirParadas() {
      return autenticado() && !esMantCompartido() && !soloVista() && (esAdmin() ||
        (perfiles()[request.auth.uid].permisos is list && 'corregir_paradas' in perfiles()[request.auth.uid].permisos));
    }

    // Constancia de una corrección de asignación de paradas (quitar de la marca, reasignar de marca o de línea): UN documento por operación, con id = identificador de la operación.
    //  - La parada original NO se borra: aquí solo queda quién, cuándo (hora del servidor), qué paradas y minutos, de dónde a dónde y por qué (motivo de 5 o más caracteres).
    //  - Se crea una sola vez (crear con un id que ya existe falla: un reintento o un doble clic no la repite) y nunca se edita ni se borra.
    //  - Leen todos los usuarios con sesión salvo la cuenta compartida de Mantenimiento.
    match /correccionesParada/{id} {
      allow read: if autenticado() && !esMantCompartido();
      allow create: if puedeCorregirParadas()
        && request.resource.data.keys().hasAll(['id', 'accion', 'linea', 'fecha', 'turno', 'paradas', 'desde', 'hacia', 'minutos', 'motivo', 'cerrado', 'version', 'actualizadaUid', 'actualizadaEn'])
        && request.resource.data.keys().hasOnly(['id', 'accion', 'linea', 'fecha', 'turno', 'paradas', 'desde', 'hacia', 'minutos', 'motivo', 'cerrado', 'version', 'actualizadaUid', 'actualizadaPor', 'actualizadaEn'])
        && request.resource.data.id == id
        && request.resource.data.accion in ['QUITAR_MARCA', 'REASIGNAR_MARCA', 'REASIGNAR_LINEA']
        && request.resource.data.linea in ['PET1', 'PET2', 'B7L', 'C20L', 'B20L', 'HIELO']
        && request.resource.data.fecha is string && request.resource.data.fecha.size() == 10
        && request.resource.data.turno in ['DÍA', 'NOCHE']
        && request.resource.data.paradas is list && request.resource.data.paradas.size() > 0 && request.resource.data.paradas.size() <= 50
        && request.resource.data.motivo is string && request.resource.data.motivo.size() >= 5 && request.resource.data.motivo.size() <= 200
        && request.resource.data.version == 1
        && request.resource.data.actualizadaUid == request.auth.uid
        && request.resource.data.actualizadaEn == request.time;
      allow update, delete: if false;
    }

    // Reclasificación de paradas ya registradas (Avance/Cierre o Nuevo registro): UN documento por parada, con id = id de la parada (con «/» cambiado por «_»).
    //  - La parada original NO se toca: aquí solo se guarda la causa, el motivo, el detalle y el equipo que le asigna Mantenimiento (los minutos no cambian).
    //  - Escriben el Supervisor de Mantenimiento y el Administrador (esSupervisorMtto: no la cuenta compartida, ni jefatura, ni cuentas de solo consulta).
    //    Leen quienes ven la bitácora de Mantenimiento.
    //  - Se crea con version 1, su uid, la hora del servidor y UNA entrada de historial. Cada cambio sube la version en 1, lleva uid y hora del servidor y AGREGA
    //    exactamente una entrada al final de la lista historial; las entradas anteriores no cambian (el historial nunca se acorta ni se edita). Nadie borra.
    //  - Campos de origen (id, paradaId, origen, linea, fecha, turno, descripcionOriginal, minutos) no se cambian nunca.
    match /reclasificacionesParada/{id} {
      allow read: if puedeLeerBitacoraMtto();
      allow create: if esSupervisorMtto()
        && request.resource.data.keys().hasAll(['id', 'paradaId', 'origen', 'linea', 'fecha', 'causa', 'motivo', 'version', 'historial', 'actualizadaUid', 'actualizadaEn'])
        && request.resource.data.keys().hasOnly(['id', 'paradaId', 'origen', 'linea', 'fecha', 'turno', 'descripcionOriginal', 'minutos', 'causa', 'motivo', 'detalle',
             'equipo', 'version', 'historial', 'actualizadaUid', 'actualizadaPor', 'actualizadaEn'])
        && request.resource.data.id == id
        && request.resource.data.origen in ['AVANCE', 'REGISTRO']
        && request.resource.data.linea in ['PET1', 'PET2', 'B7L', 'C20L', 'B20L']
        && request.resource.data.fecha is string && request.resource.data.fecha.size() == 10
        && request.resource.data.causa is string && request.resource.data.causa.size() > 0
        && request.resource.data.motivo is string && request.resource.data.motivo.size() > 0
        && request.resource.data.version == 1
        && request.resource.data.actualizadaUid == request.auth.uid
        && request.resource.data.actualizadaEn == request.time
        && request.resource.data.historial is list && request.resource.data.historial.size() == 1;
      allow update: if esSupervisorMtto()
        && request.resource.data.keys().hasOnly(['id', 'paradaId', 'origen', 'linea', 'fecha', 'turno', 'descripcionOriginal', 'minutos', 'causa', 'motivo', 'detalle',
             'equipo', 'version', 'historial', 'actualizadaUid', 'actualizadaPor', 'actualizadaEn'])
        && request.resource.data.version == resource.data.version + 1
        && request.resource.data.actualizadaUid == request.auth.uid
        && request.resource.data.actualizadaEn == request.time
        && request.resource.data.causa is string && request.resource.data.causa.size() > 0
        && request.resource.data.motivo is string && request.resource.data.motivo.size() > 0
        && request.resource.data.historial is list
        && request.resource.data.historial.size() == resource.data.historial.size() + 1
        && request.resource.data.historial[0:resource.data.historial.size()] == resource.data.historial
        && !request.resource.data.diff(resource.data).affectedKeys()
             .hasAny(['id', 'paradaId', 'origen', 'linea', 'fecha', 'turno', 'descripcionOriginal', 'minutos']);
      allow delete: if false;
    }

    // Entrega de turno entre supervisores: UN documento por fecha operativa y turno SALIENTE (id = fecha + '_' + turno).
    //  - Lo leen todos los usuarios con sesión, salvo la cuenta compartida de Mantenimiento (jefatura y gerencia en modo consulta).
    //  - Crea quien opera Producción (puedeOperarProduccion), con su uid y la hora del servidor; crear con un id existente falla.
    //  - Mientras esté ENTREGADA, solo quien la entregó corrige resumen y novedades (más actualizadaEn y actualizaciones).
    //  - Recibir: otro usuario (distinto del que entregó) pasa estado a RECIBIDA y escribe solo recibeUid, recibePor, recibeNombre,
    //    recibidaEn (hora del servidor) y observacionRecibe (opcional). Después de RECIBIDA nadie la cambia. Nunca se borra.
    match /entregasTurno/{id} {
      allow read: if autenticado() && !esMantCompartido();
      allow create: if puedeOperarProduccion()
        && request.resource.data.keys().hasAll(['fecha', 'turno', 'estado', 'resumen', 'novedades', 'entregaUid', 'entregaPor',
             'entregaNombre', 'entregadaEn'])
        && request.resource.data.keys().hasOnly(['fecha', 'turno', 'estado', 'resumen', 'novedades', 'entregaUid', 'entregaPor',
             'entregaNombre', 'entregadaEn'])
        && request.resource.data.turno in ['DIA', 'INTERMEDIO', 'NOCHE']
        && id == request.resource.data.fecha + '_' + request.resource.data.turno
        && request.resource.data.estado == 'ENTREGADA'
        && request.resource.data.novedades is list && request.resource.data.novedades.size() <= 30
        && request.resource.data.entregaUid == request.auth.uid
        && request.resource.data.entregadaEn == request.time;
      allow update: if puedeOperarProduccion()
        && resource.data.estado == 'ENTREGADA'
        && (
          // Corregir: solo quien entregó, solo resumen y novedades.
          (resource.data.entregaUid == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys()
                 .hasOnly(['resumen', 'novedades', 'actualizadaEn', 'actualizaciones'])
            && request.resource.data.novedades is list && request.resource.data.novedades.size() <= 30
            && request.resource.data.actualizadaEn == request.time)
          ||
          // Recibir: otro usuario, solo los campos de recepción.
          (resource.data.entregaUid != request.auth.uid
            && request.resource.data.estado == 'RECIBIDA'
            && request.resource.data.diff(resource.data).affectedKeys()
                 .hasOnly(['estado', 'recibeUid', 'recibePor', 'recibeNombre', 'recibidaEn', 'observacionRecibe'])
            && request.resource.data.diff(resource.data).affectedKeys()
                 .hasAll(['estado', 'recibeUid', 'recibePor', 'recibeNombre', 'recibidaEn'])
            && request.resource.data.recibeUid == request.auth.uid
            && request.resource.data.recibidaEn == request.time)
        );
      allow delete: if false;
    }

    // Avances y cierres de turno: UN documento por snapshot (id = el mismo id del snapshot, p. ej. 2026-10-10|DÍA|CIERRE).
    // Antes vivían en sync/avancesTurno.items y ese documento se acercaba al límite de 1 MiB de Firestore.
    //  - Lo leen los mismos que leen sync/avancesTurno (usuario con sesión; la cuenta compartida solo si ya lo leía: sí).
    //  - Crea quien puede generar el cierre (puedeGenerarResumen), con su uid y la hora del servidor; crear con un id existente
    //    falla (un cierre/avance ya generado no se pisa). El contenido del snapshot conserva los mismos campos de siempre.
    //  - Solo se actualizan copiadoEn y copiadoPor. Nadie borra.
    match /avancesTurnoItems/{id} {
      allow read: if autenticado();
      allow create: if puedeGenerarResumen()
        && request.resource.data.keys().hasAll(['id', 'fecha', 'turno', 'tipo', 'creadoPorUid', 'creadoEn'])
        && request.resource.data.id == id
        && request.resource.data.fecha is string
        && request.resource.data.tipo in ['AVANCE', 'CIERRE']
        && request.resource.data.creadoPorUid == request.auth.uid
        && request.resource.data.creadoEn == request.time;
      allow update: if puedeGenerarResumen()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['copiadoEn', 'copiadoPor']);
      allow delete: if false;
    }

    // Compromiso original de la programación (Planificación, etapa 2): UN documento por bloque (id = AAAA-MM-DD_DIA o _NOCHE) con el plan
    // vigente al INICIAR el bloque. Lo crea quien planifica, una sola vez, justo antes del primer cambio de plan posterior al inicio.
    //  - Leen todos los usuarios con sesión (menos la cuenta compartida). Nunca se edita ni se borra.
    match /compromisosProgramacion/{id} {
      allow read: if autenticado() && !esMantCompartido();
      allow create: if puedePlanificar() && !soloVista()
        && request.resource.data.keys().hasAll(['fecha', 'bloque', 'filas', 'inicioMs', 'creadoEn', 'uid'])
        && request.resource.data.bloque in ['DÍA', 'NOCHE']
        && request.resource.data.filas is list
        && id == request.resource.data.fecha + '_' + request.resource.data.bloque
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.creadoEn == request.time;
      allow update, delete: if false;
    }

    // Distribución de personal por línea, por fecha operativa y bloque (documento = AAAA-MM-DD_DIA o _NOCHE).
    //  - El documento solo guarda la versión (cada guardado sube la versión en 1: dos personas no se pisan).
    //  - Los eventos (historial) solo se crean: uid propio y hora del servidor; nunca se editan ni se borran.
    //  - Leen todos los usuarios con sesión (menos la cuenta compartida); escribe quien tiene permiso de distribuir.
    match /distribucionPersonal/{id} {
      allow read: if autenticado() && !esMantCompartido();
      allow create: if puedeDistribuirPersonal()
        && request.resource.data.version == 1
        && request.resource.data.actualizadoUid == request.auth.uid
        && request.resource.data.actualizadoEn == request.time;
      allow update: if puedeDistribuirPersonal()
        && request.resource.data.version == resource.data.version + 1
        && request.resource.data.actualizadoUid == request.auth.uid
        && request.resource.data.actualizadoEn == request.time;
      allow delete: if false;
      match /eventos/{opId} {
        allow read: if autenticado() && !esMantCompartido();
        allow create: if puedeDistribuirPersonal()
          && request.resource.data.opId == opId
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.guardadoEn == request.time
          && request.resource.data.keys().hasAll(['opId', 'version', 'desdeMs', 'lineas', 'uid', 'guardadoEn']);
        allow update, delete: if false;
      }
    }

    // Historial de cambios de la programación: solo se agrega, nunca se edita ni se borra.
    // Cada evento lleva el uid de quien lo escribe y la hora del servidor. Lo leen quienes planifican y la jefatura.
    match /historialProgramacion/{id} {
      allow read: if puedePlanificar() || (autenticado() && !esMantCompartido() && esJefatura());
      allow create: if autenticado() && !esMantCompartido() && !soloVista()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.timestamp == request.time
        && request.resource.data.keys().hasAll(['uid', 'timestamp', 'accion', 'linea', 'fecha', 'turno'])
        && request.resource.data.accion in ['CREACION', 'EDICION', 'ELIMINACION', 'COPIA', 'IMPORTACION', 'APROBACION', 'RECHAZO_SOLICITUD'];
      allow update, delete: if false;
    }

    // Solicitudes de programación (cambio de cantidad o de producto, o programación nueva).
    //  - Cualquier usuario con sesión (menos la cuenta compartida) crea la suya: su uid, PENDIENTE, motivo y hora del servidor.
    //  - Cada quien lee las suyas; quien planifica lee todas y solo puede resolverlas (APROBADA o RECHAZADA) con comentario.
    //  - Nadie las borra.
    match /solicitudesProgramacion/{id} {
      allow read: if autenticado() && !esMantCompartido() && (puedePlanificar() || resource.data.uid == request.auth.uid);
      allow create: if autenticado() && !esMantCompartido() && !soloVista()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.estado == 'PENDIENTE'
        && request.resource.data.creadoEn == request.time
        && request.resource.data.motivo is string && request.resource.data.motivo.size() >= 5
        && request.resource.data.tipo in ['CAMBIO_CANTIDAD', 'CAMBIO_PRODUCTO', 'NUEVA_PROGRAMACION'];
      allow update: if puedePlanificar()
        && resource.data.estado == 'PENDIENTE'
        && request.resource.data.estado in ['APROBADA', 'RECHAZADA']
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['estado', 'comentario', 'resueltaPor', 'resueltaUid', 'resueltaEn'])
        && request.resource.data.resueltaUid == request.auth.uid
        && request.resource.data.resueltaEn == request.time;
      allow delete: if false;
    }

    // VERSIÓN ANTERIOR de los valores económicos (margen, costos y meta). Cerrada: solo Gerencia la lee, para migrarla a valoresUnitarios.
    // Nadie la escribe. Una vez migrada se borra desde Firebase Console.
    match /configEconomica/{doc} {
      allow read: if ecoGestionar() && doc in ['margenes', 'costos', 'general'];
      allow write: if false;
    }

    // Valores unitarios (margen por línea + marca + presentación, costo por insumo y meta). SOLO Gerencia (por rol) lee y escribe.
    // Jefatura, Administrador, supervisores y demás reciben permission-denied. No se borra: se usa activo=false.
    // Cada cambio sube la versión (+1) y debe crear, en la misma transacción, su documento de historial: valoresUnitariosHistorial/{id}_{version}.
    match /valoresUnitarios/{id} {
      allow read: if ecoGestionar();
      allow create, update: if ecoGestionar() && !soloVista()
        && request.resource.data.keys().hasOnly(['tipo', 'linea', 'marca', 'presentacion', 'componente', 'unidad', 'clave',
             'valorUnitario', 'fechaVigencia', 'vigencias', 'activo', 'version', 'actualizadoPorUid', 'actualizadoPorNombre', 'fechaActualizacion'])
        && request.resource.data.tipo in ['producto', 'insumo', 'meta', 'linea']
        // Valor GENERAL por línea: id L__<clave>, línea y unidad válidas (la unidad es la que usa el motor de cálculo).
        && (request.resource.data.tipo != 'linea'
            || (request.resource.data.linea in ['PET1', 'PET2', 'B7L', 'C20L', 'B20L']
                && request.resource.data.unidad in ['Botella', 'Bidón', 'Caja']
                && id == 'L__' + request.resource.data.clave))
        && (!('fechaVigencia' in request.resource.data) || (request.resource.data.fechaVigencia is string
            && request.resource.data.fechaVigencia.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')))
        && request.resource.data.valorUnitario is number && request.resource.data.valorUnitario >= 0
        && request.resource.data.version == (resource == null ? 1 : resource.data.version + 1)
        && request.resource.data.actualizadoPorUid == request.auth.uid
        && request.resource.data.fechaActualizacion == request.time
        && existsAfter(/databases/$(database)/documents/valoresUnitariosHistorial/$(id + '_' + string(request.resource.data.version)));
      allow delete: if false;
    }

    // Historial de valores unitarios: solo se agrega (nunca se edita ni se borra); solo Gerencia lo lee y lo crea, con su UID y la hora del servidor.
    match /valoresUnitariosHistorial/{id} {
      allow read: if ecoGestionar();
      allow create: if ecoGestionar() && !soloVista()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.timestamp == request.time
        && request.resource.data.keys().hasAll(['valorId', 'version', 'valorNuevo', 'uid', 'timestamp'])
        && id == request.resource.data.valorId + '_' + string(request.resource.data.version)
        && getAfter(/databases/$(database)/documents/valoresUnitarios/$(request.resource.data.valorId)).data.version == request.resource.data.version;
      allow update, delete: if false;
    }

    // Resultados económicos autorizados (soles ya calculados, sin valores unitarios ni detalle por producto). Los escribe Gerencia
    // (su navegador los calcula); los leen Gerencia y la Jefatura con acceso económico. El Administrador, los supervisores y demás: permission-denied.
    match /resultadosEconomicos/{id} {
      allow read: if ecoVer();
      allow create, update: if ecoGestionar()
        && request.resource.data.generadoPorUid == request.auth.uid
        && request.resource.data.generadoEn == request.time;
      allow delete: if false;
    }

    // Bitácora del tareo: se puede leer y agregar, nunca modificar ni borrar.
    // Al crear, el evento debe llevar el uid de quien lo escribe (no se puede
    // registrar a nombre de otro) y la hora del servidor (no la del dispositivo).
    match /auditoriaTareos/{evento} {
      allow read: if autenticado() && !esMantCompartido();
      allow create: if autenticado() && !esMantCompartido() && !soloVista()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.timestamp == request.time;
      allow update, delete: if false;
    }

    // Motivos de no producción por línea (INFORMATIVO: no es una parada oficial ni cambia tiempos, ratio o producción).
    // Un documento por fecha operativa + bloque + línea (id = AAAA-MM-DD_DIA|NOCHE_LINEA): no hay dos vigentes de la misma situación.
    //  - Leen todos los usuarios con sesión (menos la cuenta compartida de Mantenimiento), incluida la Jefatura en solo consulta.
    //  - Escribe quien opera Producción (puedeOperarProduccion: ya excluye soloVista y la cuenta compartida), con su uid y la hora del servidor.
    //  - El historial solo crece (nunca se acorta) y nadie borra incidencias.
    match /incidenciasNoProduccion/{id} {
      allow read: if autenticado() && !esMantCompartido();
      allow create, update: if puedeOperarProduccion()
        && request.resource.data.keys().hasOnly(['linea', 'fecha', 'bloque', 'estado', 'motivo', 'descripcion', 'registradoPor',
             'registradoUid', 'registradoEn', 'resueltoEn', 'resueltoPor', 'resueltoPorEvento', 'historial', 'actualizadoUid', 'actualizadoEn'])
        && request.resource.data.keys().hasAll(['linea', 'fecha', 'bloque', 'estado', 'motivo', 'registradoUid', 'registradoEn',
             'historial', 'actualizadoUid', 'actualizadoEn'])
        && request.resource.data.linea in ['PET1', 'PET2', 'B7L', 'C20L', 'B20L']
        && request.resource.data.bloque in ['DIA_INTERMEDIO', 'NOCHE']
        && request.resource.data.estado in ['VIGENTE', 'RESUELTA']
        && id == request.resource.data.fecha + '_' + (request.resource.data.bloque == 'NOCHE' ? 'NOCHE' : 'DIA') + '_' + request.resource.data.linea
        && request.resource.data.motivo in ['Falta de personal', 'Falla de máquina', 'Falta de insumos', 'Mantenimiento', 'Limpieza',
             'Cambio de formato', 'Otra causa']
        && (request.resource.data.motivo != 'Otra causa'
            || (request.resource.data.descripcion is string && request.resource.data.descripcion.size() >= 3))
        && request.resource.data.actualizadoUid == request.auth.uid
        && request.resource.data.actualizadoEn == request.time
        && request.resource.data.historial is list && request.resource.data.historial.size() >= 1
        && (resource == null || request.resource.data.historial.size() >= resource.data.historial.size());
      allow delete: if false;
    }

    // Tareo en un documento por tareo (TAREO_POR_DOCUMENTO en js/nucleo/01-config.js; js/personal/13n-tareo-nube.js).
    //  - Leen los mismos que leen sync/tareos: todo usuario con sesión (incluida la cuenta compartida de Mantenimiento).
    //  - Escriben los mismos que escriben sync/tareos: con sesión, sin modo solo consulta y no la cuenta compartida. El control por área
    //    (Producción / Mantenimiento) sigue en la aplicación, igual que hoy con sync/tareos.
    //  - El id del documento es el id del tareo. No se borra: eliminar deja eliminado:true con sus datos (se puede recuperar).
    match /tareos/{id} {
      allow read: if autenticado();
      allow create, update: if autenticado() && !esMantCompartido() && !soloVista()
        && request.resource.data.id == id;
      allow delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
